## ローター積
$$\begin{aligned}
	R_{\mathrm{a}}R_{\mathrm{b}} &= \left(c_{\mathrm{a}} + s_{\mathrm{a}}\mathrm{A}\right)\left(c_{\mathrm{b}} + s_{\mathrm{b}}\mathrm{B}\right) \\
	&= c_{\mathrm{a}}c_{\mathrm{b}} + c_{\mathrm{a}}s_{\mathrm{b}}\mathrm{B} + s_{\mathrm{a}}c_{\mathrm{b}}\mathrm{A} + s_{\mathrm{a}}s_{\mathrm{b}}\mathrm{A}\mathrm{B}
\end{aligned}$$
$$\begin{aligned}
	&I=e_1e_2e_3\text{において}\quad Ie_1=-e_2e_3, Ie_2=-e_3e_1, Ie_3=-e_1e_2\text{となるので}
\end{aligned}$$
$$\begin{aligned}
	\mathrm{A}\mathrm{B} &= -\mathrm{A}\cdot_E\mathrm{B} - I\left(\mathrm{A}\times\mathrm{B}\right) \\
	&= -\mathrm{A}_{2,3}\mathrm{B}_{2,3} - \mathrm{A}_{3,1}\mathrm{B}_{3,1} - \mathrm{A}_{1,2}\mathrm{B}_{1,2} \\
	&+ \left(\mathrm{A}_{3,1}\mathrm{B}_{1,2} - \mathrm{A}_{1,2}\mathrm{B}_{3,1}\right)e_2e_3 \\
	&+ \left(\mathrm{A}_{1,2}\mathrm{B}_{2,3} - \mathrm{A}_{2,3}\mathrm{B}_{1,2}\right)e_3e_1 \\
	&+ \left(\mathrm{A}_{2,3}\mathrm{B}_{3,1} - \mathrm{A}_{3,1}\mathrm{B}_{2,3}\right)e_1e_2
\end{aligned}$$
$$\begin{aligned}
	\mathrm{S} &= -\left(\mathrm{A}_{2,3}\mathrm{B}_{2,3} + \mathrm{A}_{3,1}\mathrm{B}_{3,1} + \mathrm{A}_{1,2}\mathrm{B}_{1,2}\right) \\
	\mathrm{W}_{2,3} &= \mathrm{A}_{3,1}\mathrm{B}_{1,2} - \mathrm{A}_{1,2}\mathrm{B}_{3,1} \\
	\mathrm{W}_{3,1} &= \mathrm{A}_{1,2}\mathrm{B}_{2,3} - \mathrm{A}_{2,3}\mathrm{B}_{1,2} \\
	\mathrm{W}_{1,2} &= \mathrm{A}_{2,3}\mathrm{B}_{3,1} - \mathrm{A}_{3,1}\mathrm{B}_{2,3}
\end{aligned}$$
$$\begin{aligned}
	R_{\mathrm{a}}R_{\mathrm{b}} &= \left(c_{\mathrm{a}}c_{\mathrm{b}} + s_{\mathrm{a}}s_{\mathrm{b}}\mathrm{S}\right) \\
	&+ \left(c_{\mathrm{a}}s_{\mathrm{b}}\mathrm{B}_{2,3} + s_{\mathrm{a}}c_{\mathrm{b}}\mathrm{A}_{2,3} + s_{\mathrm{a}}s_{\mathrm{b}}\mathrm{W}_{2,3}\right)e_2e_3 \\
	&+ \left(c_{\mathrm{a}}s_{\mathrm{b}}\mathrm{B}_{3,1} + s_{\mathrm{a}}c_{\mathrm{b}}\mathrm{A}_{3,1} + s_{\mathrm{a}}s_{\mathrm{b}}\mathrm{W}_{3,1}\right)e_3e_1 \\
	&+ \left(c_{\mathrm{a}}s_{\mathrm{b}}\mathrm{B}_{1,2} + s_{\mathrm{a}}c_{\mathrm{b}}\mathrm{A}_{1,2} + s_{\mathrm{a}}s_{\mathrm{b}}\mathrm{W}_{1,2}\right)e_1e_2
\end{aligned}$$

## 双ベクトル $(\mathrm{B})$ を基にローター $(\Delta R)$ を指数写像によって作成
$$\begin{aligned}
	\Omega = \mathrm{B}_{2,3}e_2e_3 + \mathrm{B}_{3,1}e_3e_1 + \mathrm{B}_{1,2}e_1e_2
\end{aligned}$$
$$\begin{aligned}
	\Omega^2 &= -\left(\mathrm{B}_{2,3}\right)^2 - \left(\mathrm{B}_{3,1}\right)^2 - \left(\mathrm{B}_{1,2}\right)^2 \\
	&\text{交差項が消える過程は別資料を参照} \\
	\omega^2 &= \left(\mathrm{B}_{2,3}\right)^2 + \left(\mathrm{B}_{3,1}\right)^2 + \left(\mathrm{B}_{1,2}\right)^2 \\
	&\therefore\Omega^2 = -\omega^2 \Longrightarrow \mathrm{\widehat{B}} = \frac{\Omega}{\omega}\quad\left(\mathrm{\widehat{B}}^2 = -1\right)
\end{aligned}$$
$$\begin{aligned}
	&x = \omega\theta,\quad\theta = \frac{\Delta t}{2} \\
	&\Delta R = e^{\textstyle{\mathrm{\widehat{B}}x}} = \cos{x} + \mathrm{\widehat{B}}\sin{x}
\end{aligned}$$
$$\begin{aligned}
	&\text{ケイリー変換のパデ近似を用いる} \\
	&e^{\textstyle\mathrm{\widehat{B}}x} = \frac{1 + \mathrm{\widehat{B}}\alpha}{1 - \mathrm{\widehat{B}}\alpha}\quad\left(\alpha = \tan\frac{x}{2}\right),\quad\tan{\frac{x}{2}} \approx \frac{6x}{12 - x^2}
\end{aligned}$$
$$\begin{aligned}
	&\therefore\frac{\left(12 - x^2\right) + \mathrm{\widehat{B}}6x}{\left(12 - x^2\right) - \mathrm{\widehat{B}}6x} \\
	&= \frac{\left(144 - 60x^2 + x^4\right) + \mathrm{\widehat{B}}\left(144x - 12x^3\right)}{144 + 12x^2 + x^4}
\end{aligned}$$
$$\begin{aligned}
	D &= \frac{1}{144 + 12x^2 + x^4} \\
	\cos{x} &\approx D\left(144 - 60x^2 + x^4\right) \\
	\mathrm{\widehat{B}}\sin{x} &\approx \mathrm{\widehat{B}}D\left(144x - 12x^3\right)
\end{aligned}$$
$$\begin{aligned}
	&x = \omega\theta,\mathrm{\widehat{B}} = \frac{\Omega}{\omega}\quad\therefore\mathrm{\widehat{B}}x = \Omega\theta\text{なので}
\end{aligned}$$
$$\begin{aligned}
	&\therefore\mathrm{\widehat{B}}D\left(144x - 12x^3\right) \Longrightarrow \Omega\theta D\left(144 - 12x^2\right) \\
	&\text{この通り変形することで}(\omega\rightarrow 0){をサポートする} \\
	&\therefore\Delta R \approx D\left(144 - 60x^2 + x^4\right) + \Omega\theta D\left(144 - 12x^2\right)
\end{aligned}$$

## 現在の姿勢における鉛直方向と加速度センサー $(a_{\mathrm{sens}})$ による比較ベクトル $(v_{\mathrm{ref}}, v_{\mathrm{target}})$ の取得

	v_{\mathrm{target}} &= \frac{a_{\mathrm{sens}}}{\|a_{\mathrm{sens}}\|_2}\quad\cdots\text{正規化加速度センサー値} \\
	v_{\mathrm{ref}} &= \widetilde{R}\left(e_3\right)R\quad\cdots\text{現在の姿勢における鉛直方向}
\end{aligned}$$

## 方位センサー $(m_{\mathrm{sens}})$ を基にした現在の姿勢における水平面での回転と正規化方位センサー値による比較ベクトル $(v_{\mathrm{ref}}, v_{\mathrm{target}})$ の取得
$$\begin{aligned}
	v_{\mathrm{target}} &= \frac{m_{\mathrm{sens}}}{\|m_{\mathrm{sens}}\|_2}\quad\cdots\text{正規化方位センサー値} \\

## 比較ベクトル $(v_{\mathrm{ref}}, v_{\mathrm{target}})$ から姿勢補正双ベクトル $(\Gamma)$ を作成
$$\begin{aligned}
	a &= {v_{\mathrm{target}}}_1e_1 + {v_{\mathrm{target}}}_2e_2 + {v_{\mathrm{target}}}_3e_3 \\
	b &= {v_{\mathrm{ref}}}_1e_1 + {v_{\mathrm{ref}}}_2e_2 + {v_{\mathrm{ref}}}_3e_3
\end{aligned}$$
$$\begin{aligned}
	\Gamma &= a\wedge b \\
	&= \left(a_2b_3 - a_3b_2\right)e_2e_3 \\
	&+ \left(a_3b_1 - a_1b_3\right)e_3e_1 \\
	&+ \left(a_1b_2 - a_2b_1\right)e_1e_2
\end{aligned}$$
$$\begin{aligned}
	&\text{厳密な誤差回転量は以下のとおりであるが小角近似を用いる}
\end{aligned}$$
$$\begin{aligned}
	\phi &= \mathrm{atan2}\left(\|a\wedge b\|_2, a\cdot_E b\right) \\
	\mathrm{\widehat{B}} &= \frac{a\wedge b}{\|a\wedge b\|_2} \\
	&e^{\textstyle{\mathrm{\widehat{B}}}\dfrac{\phi}{2}}
\end{aligned}$$

## 姿勢補正双ベクトル $(\Gamma)$ を蓄積してドリフト補正双ベクトル $(\mathrm{E})$ を取得
$$\begin{aligned}
	&\mathrm{E} = \int 2\pi f_c\left(\Gamma - \mathrm{E}\right)dt \Longrightarrow \frac{d\mathrm{E}}{dt} = 2\pi f_c\left(\Gamma - \mathrm{E}\right)
\end{aligned}$$
$$\begin{aligned}
	&\mathrm{E}_{n+1} = \mathrm{E}_n + \mathrm{E}_{\mathrm{damp}}\left(\Gamma_n - \mathrm{E}_n\right) \\
	&\mathrm{E}_{\mathrm{damp}} = 1-e^{\textstyle-2\pi f_c\Delta t}
\end{aligned}$$
$$\begin{aligned}
	&x = 2\pi f_c\Delta t \\
	&e^{\textstyle-x} \approx \frac{12+x^2-6x}{12+x^2+6x} \\
	&\mathrm{E}_{\mathrm{damp}} \approx 1-\frac{12+x^2-6x}{12+x^2+6x} = \frac{12x}{12+x^2+6x}
\end{aligned}$$
