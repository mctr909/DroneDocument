> # 定義  
>> ## 加法  
>> ### 加法交換律  
>> $$e_i+e_j=e_j+e_i$$
>>  
>> ### 加法結合律  
>> $$(e_i+e_j)+e_k=e_i+(e_j+e_k)$$
>>  
>> ### 加法単位元  
>> $$\begin{aligned}
	&e_i+0=e_i \\
	&0+e_i=e_i
\end{aligned}$$
>>  
>> ### 加法逆元  
>> $$\begin{aligned}
	&e_i+(-e_i)=0 \\
	&-e_i+e_i=0
\end{aligned}$$
>  
>> ## 乗法  
>> ### 乗法反交換律  
>> $$e_ie_j=-e_je_i\quad\left(i\neq j\right)$$
>>  
>> ### 平方 $({e_i}^2)$ と符号数 $(\lambda_i)$  
>> $${e_i}^2=\lambda_i\quad\left(\lambda_i\in\{-1,0,1\}\right)$$
>>  
>> ### 乗法結合律  
>> $$(e_ie_j)e_k=e_i(e_je_k)$$
>>  
>> ### 乗法単位元  
>> $$\begin{aligned}
	&1e_i=e_i \\
	&e_i1=e_i
\end{aligned}$$
>>  
>> ### 乗法逆元  
>> $$\begin{aligned}
	&{e_i}^{-1}=\frac{e_i}{{e_i}^2}=\frac{e_i}{\lambda_i}\quad\left(\lambda_i\in\{-1,1\}\right) \\
	&e_i{e_i}^{-1}=1 \\
	&{e_i}^{-1}e_i=1
\end{aligned}$$
>  
>> ## 分配律  
>> $$\begin{aligned}
	&e_i(e_j+e_k) = e_ie_j+e_ie_k \\
	&(e_i+e_j)e_k = e_ie_k+e_je_k
\end{aligned}$$
>  
>> ## 乗法反交換律と平方 $({e_i}^2)$ と符号数 $(\lambda_i)$のまとめ  
>> $$\begin{matrix}
	\text{例として基底ベクトルが3つ}\left(e_1,e_2,e_3\right)\text{の場合、以下の様にまとめられる} \\
	{e_1}^2=\lambda_1,{e_2}^2=\lambda_2,{e_3}^2=\lambda_3 \\
	\\
	\begin{aligned}
		e_ie_j + e_je_i &= 2g_{ij} \\
		g &= \mathrm{diag}\left(\lambda_1,\lambda_2,\lambda_3\right)
	\end{aligned} \\
	\\
	\mathrm{diag}\left(\lambda_1,\lambda_2,\lambda_3\right) = \begin{bmatrix}
		\lambda_1 & 0 & 0 \\
		0 & \lambda_2 & 0 \\
		0 & 0 & \lambda_3
	\end{bmatrix}
\end{matrix}$$
>  

> # 単純バイベクトル  
>> ## 基底のペア  
>> $$\begin{matrix}
	&\text{異なる2つ1組の基底の積を単純バイベクトルと呼び、} \\
	&\begin{aligned}
		&{e_p}^2={e_{p1}}^2={e_{p2}}^2=1 \\
		&{e_q}^2={e_{q1}}^2={e_{q2}}^2=-1 \\
		&{e_r}^2={e_{r1}}^2={e_{r2}}^2=0
	\end{aligned} \\
	&\text{と置いた時に組み合わさる種類としては以下のとおり}
\end{matrix}$$
>> $$\begin{matrix}
	\text{同種} & \text{同種(反交換)} & \text{異種} & \text{異種(反交換)} \\
	\hline
	e_{p1}e_{p2} & e_{p2}e_{p1} & e_{p}e_{q} & e_{q}e_{p} \\
	e_{q1}e_{q2} & e_{q2}e_{q1} & e_{p}e_{r} & e_{r}e_{p} \\
	e_{r1}e_{r2} & e_{r2}e_{r1} & e_{q}e_{r} & e_{r}e_{q}
\end{matrix}$$
>  
>> ## 単純バイベクトルの2乗  
>> $$\begin{matrix}
	\begin{aligned}
		&\text{同種の基底の関係} \\
		(e_{p1}e_{p2})^2 &= (e_{p1}e_{p2})(e_{p1}e_{p2}) \\
		&= e_{p1}(e_{p2}e_{p1})e_{p2} \\
		&= e_{p1}(-e_{p1}e_{p2})e_{p2} \\
		&= -e_{p1}e_{p1}e_{p2}e_{p2} \\
		&= -(e_{p1}e_{p1})(e_{p2}e_{p2}) \\
		&= -(1)(1) \\
		&= -1 \\
		(e_{q1}e_{q2})^2 &= (e_{q1}e_{q2})(e_{q1}e_{q2}) \\
		&= e_{q1}(e_{q2}e_{q1})e_{q2} \\
		&= e_{q1}(-e_{q1}e_{q2})e_{q2} \\
		&= -e_{q1}e_{q1}e_{q2}e_{q2} \\
		&= -(e_{q1}e_{q1})(e_{q2}e_{q2}) \\
		&= -(-1)(-1) \\
		&= -1 \\
		(e_{r1}e_{r2})^2 &= (e_{r1}e_{r2})(e_{r1}e_{r2}) \\
		&= e_{r1}(e_{r2}e_{r1})e_{r2} \\
		&= e_{r1}(-e_{r1}e_{r2})e_{r2} \\
		&= -e_{r1}e_{r1}e_{r2}e_{r2} \\
		&= -(e_{r1}e_{r1})(e_{r2}e_{r2}) \\
		&= -(0)(0) \\
		&= 0
	\end{aligned} &\quad
	\begin{aligned}
		&\text{異種の基底の関係} \\
		(e_{p}e_{q})^2 &= (e_{p}e_{q})(e_{p}e_{q}) \\
		&= e_{p}(e_{q}e_{p})e_{q} \\
		&= e_{p}(-e_{p}e_{q})e_{q} \\
		&= -e_{p}e_{p}e_{q}e_{q} \\
		&= -(e_{p}e_{p})(e_{q}e_{q}) \\
		&= -(1)(-1) \\
		&= 1 \\
		(e_{p}e_{r})^2 &= (e_{p}e_{r})(e_{p}e_{r}) \\
		&= e_{p}(e_{r}e_{p})e_{r} \\
		&= e_{p}(-e_{p}e_{r})e_{r} \\
		&= -e_{p}e_{p}e_{r}e_{r} \\
		&= -(e_{p}e_{p})(e_{r}e_{r}) \\
		&= -(1)(0) \\
		&= 0 \\
		(e_{q}e_{r})^2 &= (e_{q}e_{r})(e_{q}e_{r}) \\
		&= e_{q}(e_{r}e_{q})e_{r} \\
		&= e_{q}(-e_{q}e_{r})e_{r} \\
		&= -e_{q}e_{q}e_{r}e_{r} \\
		&= -(e_{q}e_{q})(e_{r}e_{r}) \\
		&= -(-1)(0) \\
		&= 0
	\end{aligned}
\end{matrix}$$
>> $$\begin{aligned}
	\text{反交換関係にある組は省略}
\end{aligned}$$
>> $$\begin{aligned}
	&\text{2乗すると符号数}\left(\lambda\in\{-1,0,1\}\right)\text{となる}
\end{aligned}$$
>  
>> ## 【まとめ】  
>> |基底の組み合わせ|単純バイベクトル|単純バイベクトル(反交換)|2乗|
>> |-|-|-|-|
>> |$e_r$を含まない同種|$(e_{p1}e_{p2}),(e_{q1}e_{q2})$|$(e_{p2}e_{p1}),(e_{q2}e_{q1})$|$\lambda=-1$|
>> |$e_r$を含む|$(e_{r1}e_{r2}),(e_{p}e_{r}),(e_{q}e_{r})$|$(e_{r2}e_{r1}),(e_{r}e_{p}),(e_{r}e_{q})$|$\lambda=0$|
>> |$e_r$を含まない異種|$(e_{p}e_{q})$|$(e_{q}e_{p})$|$\lambda=1$|
>  

> # バイベクトル  
>> ## 線形結合  
>> $$\begin{matrix}
	&c_1,c_2,c_3\quad\text{係数}\in\mathbb{R} \\
	&B_1,B_2,B_3\quad\text{単純バイベクトル} \\
	&(c_1B_1 + c_2B_2 + c_3B_3)^2\text{について線形結合の様子を確認する}
\end{matrix}$$
>> $$\begin{aligned}
	(c_1B_1 + c_2B_2 + c_3B_3)^2 &= (c_1B_1 + c_2B_2 + c_3B_3)(c_1B_1 + c_2B_2 + c_3B_3)
\end{aligned}$$
>> $$\begin{aligned}
	&= (c_1B_1)^2 + c_1B_1c_2B_2 + c_1B_1c_3B_3\quad\text{まずは素直に展開} \\
	&+ c_2B_2c_1B_1 + (c_2B_2)^2 + c_2B_2c_3B_3 \\
	&+ c_3B_3c_1B_1 + c_3B_3c_2B_2 + (c_3B_3)^2 \\
\end{aligned}$$
>> $$\begin{aligned}
	&= (c_1B_1)^2 + c_1c_2B_1B_2 + c_1c_3B_1B_3\quad\text{係数は可換なので順序を整理} \\
	&+ (c_2B_2)^2 + c_1c_2B_2B_1 + c_2c_3B_2B_3 \\
	&+ (c_3B_3)^2 + c_1c_3B_3B_1 + c_2c_3B_3B_2 \\
\end{aligned}$$
>> $$\begin{aligned}
	&= (c_1B_1)^2 + c_1c_2(B_1B_2 + B_2B_1) + c_1c_3(B_1B_3 + B_3B_1)\quad\text{同じ係数で括る} \\
	&+ (c_2B_2)^2 + c_2c_3(B_2B_3 + B_3B_2) \\
	&+ (c_3B_3)^2
\end{aligned}$$
>> $$\begin{aligned}
	\text{この結果から定式化すると}
\end{aligned}$$
>> $$\begin{aligned}
	\left(\sum_{i=1}^n c_iB_i\right)^2 = \underbrace{\sum_{i=1}^n{c_i}^2{B_i}^2}_{\text{対角項}}+\underbrace{\sum_{i<j}c_ic_j(B_iB_j + B_jB_i)}_{\text{交差項}}
\end{aligned}$$
>> $$\begin{aligned}
	\text{対角項はスカラーであり、交差項について調べる}
\end{aligned}$$
>  
>> ## 反交換子  
>> $$\begin{aligned}
	\lbrace{B_i,B_j}\rbrace = B_iB_j + B_jB_i
\end{aligned}$$
>  
>> ## 交換子  
>> $$\begin{aligned}
	\lbrack{B_i,B_j}\rbrack = B_iB_j - B_jB_i \\
\end{aligned}$$
>  
>> ## $B_i,B_j$ が同じペアの基底を持つ場合  
>> $$\begin{matrix}
	B_i = e_ae_b, B_j = e_ae_b\text{について確認} & B_i = e_ae_b, B_j = e_be_a\text{について確認} \\
	\begin{aligned}
		\lbrace{B_i,B_j}\rbrace &= (e_ae_b)(e_ae_b) + (e_ae_b)(e_ae_b) \\
		&= 2(e_ae_b)^2 \\
		&= -2\lambda_a\lambda_b\quad(\lambda_a = e^2_a,\lambda_b = e^2_b) \\
		\lbrack{B_i,B_j}\rbrack &= (e_ae_b)(e_ae_b) - (e_ae_b)(e_ae_b) \\
		&= 0
	\end{aligned}
	&
	\begin{aligned}
		\lbrace{B_i,B_j}\rbrace &= (e_ae_b)(-e_ae_b) + (-e_ae_b)(e_ae_b) \\
		&= -2(e_ae_b)^2 \\
		&= 2\lambda_a\lambda_b\quad(\lambda_a = e^2_a,\lambda_b = e^2_b) \\
		\lbrack{B_i,B_j}\rbrack &= (e_ae_b)(-e_ae_b) - (-e_ae_b)(e_ae_b) \\
		&= 0
	\end{aligned}
\end{matrix}$$
>  
>> ## $B_i,B_j$ が1つ共有する基底を持つ場合  
>> $$\begin{matrix}
	B_i = e_ae_b, B_j = e_be_c\text{について確認} & B_i = e_ae_b, B_j = e_ce_b\text{について確認} \\
	\begin{aligned}
		\lbrace{B_i,B_j}\rbrace &= (e_ae_b)(e_be_c) + (e_be_c)(e_ae_b) \\
		&= e_a(e_be_b)e_c + (-e_ce_b)(-e_be_a) \\
		&= {\lambda_b}e_ae_c + e_c(e_be_b)e_a \\
		&= {\lambda_b}e_ae_c + {\lambda_b}(e_ce_a) \\
		&= {\lambda_b}e_ae_c + {\lambda_b}(-e_ae_c) \\
		&= {\lambda_b}e_ae_c - {\lambda_b}e_ae_c \\
		&= 0 \\
		\lbrack{B_i,B_j}\rbrack &= (e_ae_b)(e_be_c) - (e_be_c)(e_ae_b) \\
		&= e_a(e_be_b)e_c - (-e_ce_b)(-e_be_a) \\
		&= {\lambda_b}e_ae_c - e_c(e_be_b)e_a \\
		&= {\lambda_b}e_ae_c - {\lambda_b}(e_ce_a) \\
		&= {\lambda_b}e_ae_c - {\lambda_b}(-e_ae_c) \\
		&= {\lambda_b}e_ae_c + {\lambda_b}e_ae_c \\
		&= 2{\lambda_b}e_ae_c\quad(\lambda_b = e^2_b)
	\end{aligned}
	&
	\begin{aligned}
		\lbrace{B_i,B_j}\rbrace &= (e_ae_b)(e_ce_b) + (e_ce_b)(e_ae_b) \\
		&= e_ae_b(-e_be_c) + e_ce_b(-e_be_a) \\
		&= -e_a(e_be_b)e_c - e_c(e_be_b)e_a \\
		&= -{\lambda_b}e_ae_c - {\lambda_b}(e_ce_a) \\
		&= -{\lambda_b}e_ae_c - {\lambda_b}(-e_ae_c) \\
		&= -{\lambda_b}e_ae_c + {\lambda_b}(e_ae_c) \\
		&= 0 \\
		\lbrack{B_i,B_j}\rbrack &= (e_ae_b)(e_ce_b) - (e_ce_b)(e_ae_b) \\
		&= e_ae_b(-e_be_c) - e_ce_b(-e_be_a) \\
		&= -e_a(e_be_b)e_c + e_c(e_be_b)e_a \\
		&= -{\lambda_b}e_ae_c + {\lambda_b}(e_ce_a) \\
		&= -{\lambda_b}e_ae_c + {\lambda_b}(-e_ae_c) \\
		&= -{\lambda_b}e_ae_c - {\lambda_b}(e_ae_c) \\
		&= -2{\lambda_b}e_ae_c\quad(\lambda_b = e^2_b)
	\end{aligned}
\end{matrix}$$
>  
>> ## $B_i,B_j$ が全て異なる基底を持つ場合  
>> $$\begin{matrix}
	&B_i = e_ae_b, B_j = e_ce_d\text{について確認} \\
	&\begin{aligned}
		\lbrace{B_i,B_j}\rbrace &= (e_ae_b)(e_ce_d) + (e_ce_d)(e_ae_b) \\
		&= e_ae_be_ce_d + e_ce_de_ae_b
	\end{aligned} \\
	&\begin{aligned}
		e_ce_de_ae_b &= (e_c)e_de_ae_b \\
		&= -e_de_ae_be_c \\
		&= -(e_d)e_ae_be_c \\
		&= e_ae_be_ce_d
	\end{aligned} \\
	&\therefore \lbrace{B_i,B_j}\rbrace = 2(e_ae_be_ce_d) \\
	\\
	&\begin{aligned}
		\lbrack{B_i,B_j}\rbrack &= (e_ae_b)(e_ce_d) - (e_ce_d)(e_ae_b) \\
		&= e_ae_be_ce_d - e_ce_de_ae_b \\
		&= 0
	\end{aligned}
\end{matrix}$$
>  
>> ## 【まとめ】  
>> $$\left(\sum_{i=1}^n c_iB_i\right)^2 = \underbrace{\sum_{i=1}^n{c_i}^2{B_i}^2}_{\text{対角項}} + \underbrace{\sum_{i<j}c_ic_j\lbrace{B_i,B_j}\rbrace}_{\text{交差項}}$$
>> $$\text{対角項はスカラーであり、交差項}\lbrace{B_i,B_j}\rbrace\text{について調べる}$$
>> |$B_i,B_j$|$\lbrace{B_i,B_j}\rbrace$|$\lbrack{B_i,B_j}\rbrack$|分類名|具体名|
>> |-|-|-|-|-|
>> |$(e_ae_b),(e_ae_b)$|$-2\lambda_a\lambda_b\hspace{0.5em}(\lambda_a = e^2_a,\lambda_b = e^2_b)$|$0$|グレード0|スカラー|
>> |$(e_ae_b),(e_be_a)$|$2\lambda_a\lambda_b\hspace{0.5em}(\lambda_a = e^2_a,\lambda_b = e^2_b)$|$0$|グレード0|スカラー|
>> |$(e_ae_b),(e_be_c)$|$0$|$2{\lambda_b}e_ae_c\hspace{0.5em}(\lambda_b = e^2_b)$|グレード2|バイベクトル|
>> |$(e_ae_b),(e_ce_b)$|$0$|$-2{\lambda_b}e_ae_c\hspace{0.5em}(\lambda_b = e^2_b)$|グレード2|バイベクトル|
>> |$(e_ae_b),(e_ce_d)$|$2(e_ae_be_ce_d)$|$0$|グレード4|4ベクトル|
>> 
>> $$\begin{aligned}
	&\text{交差項が消えるか、交差項がグレード0のみ残る場合において}B^2=\underbrace{\langle{B^2}\rangle_0}_{\text{グレード0}}\text{と書ける} \\
	&\text{グレード0とグレード4が残る場合において}B^2=\underbrace{\langle{B^2}\rangle_0}_{\text{グレード0}}+\underbrace{\langle{B^2}\rangle_4}_{\text{グレード4}}\text{と書ける} \\
\end{aligned}$$

> # 指数写像  
> $$\begin{matrix}
	&\text{バイベクトルが}\left(B^2=\lambda\in\mathbb{R}\right)\text{であれば、} \\
	&\text{以下のバイベクトルの累乗と指数関数のマクローリン展開を比較することで指数写像が定義できる} \\
	&\begin{aligned}
		&B^2 = \lambda \\
		&B^3 = {\lambda}B \\
		&B^4 = {\lambda}^2 \\
		&B^5 = {\lambda}^2{B} \\
		&\quad\quad\vdots
	\end{aligned} \\
	&\text{指数関数}e^\theta\text{のマクローリン展開} \\
	&\begin{aligned}
		e^\theta = \sum^\infty_{n=0}\frac{\theta^n}{n!} = 1 + \theta + \frac{\theta^2}{2!} + \frac{\theta^3}{3!} + \frac{\theta^4}{4!} + \frac{\theta^5}{5!} + \cdots
	\end{aligned}
\end{matrix}$$
>  
>> ## $(\lambda<0)$ の場合  
>> $$\begin{matrix}
	&\text{単位バイベクトル}(\mathrm{\widehat{B}})\text{を作る} \\
	&\begin{aligned}
		&\omega = \sqrt{|\lambda|} \\
		&\phi = \omega\theta \\
		&\mathrm{\widehat{B}} = \frac{B}{\omega}\hspace{0.5em}\left(\mathrm{\widehat{B}^2} = -1\right)
	\end{aligned} \\
	\\
	&\text{累乗を確認して周期性に注目} \\
	&\begin{aligned}
		&\mathrm{\widehat{B}^2} = -1 \\
		&\mathrm{\widehat{B}^3} = \mathrm{\widehat{B}^2}\mathrm{\widehat{B}} = -\mathrm{\widehat{B}} \\
		&\mathrm{\widehat{B}^4} = \left(\mathrm{\widehat{B}^2}\right)^2 = 1 \\
		&\mathrm{\widehat{B}^5} = \mathrm{\widehat{B}^4}\mathrm{\widehat{B}} = \mathrm{\widehat{B}}
	\end{aligned}
\end{matrix}$$
>> $$\begin{matrix}
	&\text{ここで指数関数のマクローリン展開の}\theta\text{に}\phi\mathrm{\widehat{B}}\text{を代入すると} \\
	&\begin{aligned}
		e^{\phi\mathrm{\widehat{B}}} &= 1 &+& \phi\mathrm{\widehat{B}} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^2}{2!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^3}{3!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^4}{4!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^5}{5!} &+& \cdots \\
		&= 1 &+& \phi\mathrm{\widehat{B}} &-& \frac{\phi^2}{2!} &-& \frac{\phi^3}{3!}\mathrm{\widehat{B}} &+& \frac{\phi^4}{4!} &+& \frac{\phi^5}{5!}\mathrm{\widehat{B}} &-& \cdots
	\end{aligned} \\
	&= \left(1 - \cfrac{\phi^2}{2!} + \cfrac{\phi^4}{4!} - \cdots\right) + \mathrm{\widehat{B}}\left(\phi - \cfrac{\phi^3}{3!} + \cfrac{\phi^5}{5!} - \cdots\right) \\
	&\text{となり} \\
	&e^{i\theta}\text{のマクローリン展開と同じ構造となる} \\
	&e^{i\theta} = \left(1 - \cfrac{\theta^2}{2!} + \cfrac{\theta^4}{4!} - \cdots\right) + i\left(\theta - \cfrac{\theta^3}{3!} + \cfrac{\theta^5}{5!} - \cdots\right)
\end{matrix}$$
>> $$\begin{aligned}
	\therefore e^{\theta{B}} &= \cos{\phi} + \mathrm{\widehat{B}}\sin{\phi}\quad\omega = \sqrt{|\lambda|}, \phi = \omega\theta, \mathrm{\widehat{B}} = \frac{B}{\omega}\hspace{0.5em}\left(\mathrm{\widehat{B}^2} = -1\right)
\end{aligned}$$
>  
>> ## $(\lambda>0)$ の場合  
>> $$\begin{matrix}
	&\text{単位バイベクトル}(\mathrm{\widehat{B}})\text{を作る} \\
	&\begin{aligned}
		&\omega = \sqrt{|\lambda|} \\
		&\phi = \omega\theta \\
		&\mathrm{\widehat{B}} = \frac{B}{\omega}\hspace{0.5em}\left(\mathrm{\widehat{B}^2} = 1\right)
	\end{aligned} \\
	\\
	&\text{累乗を確認して周期性に注目} \\
	&\begin{aligned}
		&\mathrm{\widehat{B}^2} = 1 \\
		&\mathrm{\widehat{B}^3} = \mathrm{\widehat{B}^2}\mathrm{\widehat{B}} = \mathrm{\widehat{B}} \\
		&\mathrm{\widehat{B}^4} = \left(\mathrm{\widehat{B}^2}\right)^2 = 1 \\
		&\mathrm{\widehat{B}^5} = \mathrm{\widehat{B}^4}\mathrm{\widehat{B}} = \mathrm{\widehat{B}}
	\end{aligned}
\end{matrix}$$
>> $$\begin{matrix}
	&\text{ここで指数関数のマクローリン展開の}\theta\text{に}\phi\mathrm{\widehat{B}}\text{を代入すると} \\
	&\begin{aligned}
		e^{\phi\mathrm{\widehat{B}}} &= 1 &+& \phi\mathrm{\widehat{B}} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^2}{2!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^3}{3!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^4}{4!} &+& \frac{\left(\phi\mathrm{\widehat{B}}\right)^5}{5!} &+& \cdots \\
		&= 1 &+& \phi\mathrm{\widehat{B}} &+& \frac{\phi^2}{2!} &+& \frac{\phi^3}{3!}\mathrm{\widehat{B}} &+& \frac{\phi^4}{4!} &+& \frac{\phi^5}{5!}\mathrm{\widehat{B}} &+& \cdots
	\end{aligned} \\
	&= \left(1 + \cfrac{\phi^2}{2!} + \cfrac{\phi^4}{4!} + \cdots\right) + \mathrm{\widehat{B}}\left(\phi + \cfrac{\phi^3}{3!} + \cfrac{\phi^5}{5!} + \cdots\right) \\
	&\text{となり} \\
	&\text{双曲線関数のマクローリン展開と同じ構造となる} \\
	&\begin{aligned}
		\cosh{\theta} &= 1 + \frac{\theta^2}{2!} + \frac{\theta^4}{4!} + \cdots \\
		\sinh{\theta} &= \theta + \frac{\theta^3}{3!} + \frac{\theta^5}{5!} + \cdots
	\end{aligned}
\end{matrix}$$
>> $$\begin{aligned}
	\therefore e^{\theta{B}} = \cosh{\phi} + \mathrm{\widehat{B}}\sinh{\phi}\quad\omega = \sqrt{|\lambda|}, \phi = \omega\theta, \mathrm{\widehat{B}} = \frac{B}{\omega}\hspace{0.5em}\left(\mathrm{\widehat{B}^2} = 1\right)
\end{aligned}$$
>  
>> ## $(\lambda=0)$ の場合  
>> $$\begin{matrix}
	&\text{2乗が0になるため、2乗以降はすべて0となる} \\
	&\begin{aligned}
		&B^2 = 0 \\
		&B^3 = {B^2}B = 0 \\
		&B^4 = \left(B^2\right)^2 = 0 \\
		&B^5 = {B^4}B = 0
	\end{aligned}
\end{matrix}$$
>> $$\begin{matrix}
	&\text{指数関数のマクローリン展開の}\theta\text{に}\theta B\text{を代入すると2次以降の項が消える} \\
	&\begin{aligned}
		e^{\theta B} &= 1 &+& \theta B &+& \frac{\left(\theta B\right)^2}{2!} &+& \frac{\left(\theta B\right)^3}{3!} &+& \frac{\left(\theta B\right)^4}{4!} &+& \frac{\left(\theta B\right)^5}{5!} &+& \cdots \\
		&= 1 &+& \theta B &+& 0 &+& 0 &+& 0 &+& 0 &+&\cdots \\
		&= 1 &+& \theta B
	\end{aligned}
\end{matrix}$$
>
