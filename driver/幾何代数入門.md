> # 定義  
> $$\begin{aligned}
	\text{2乗して}1\text{とする基底を}e_{p} \\  
	\text{2乗して}-1\text{とする基底を}e_{q} \\ 
	\text{2乗して}0\text{とする基底を}e_{n} \\
\end{aligned}$$
> $$\begin{aligned}
	この三種類の基底の関係を以下に示す
\end{aligned}$$
> $$\begin{aligned}
	e_{p}e_{p}+e_{p}e_{p} = &2& \quad\text{2乗} \\
	e_{p}e_{q}+e_{q}e_{p} = &0& \quad\text{反交換関係} \\
	e_{p}e_{n}+e_{n}e_{p} = &0& \quad\text{反交換関係} \\
	e_{q}e_{p}+e_{p}e_{q} = &0& \quad\text{反交換関係} \\
	e_{q}e_{q}+e_{q}e_{q} = &-2& \quad\text{2乗} \\
	e_{q}e_{n}+e_{n}e_{q} = &0& \quad\text{反交換関係} \\
	e_{n}e_{p}+e_{p}e_{n} = &0& \quad\text{反交換関係} \\
	e_{n}e_{q}+e_{q}e_{n} = &0& \quad\text{反交換関係} \\
	e_{n}e_{n}+e_{n}e_{n} = &0& \quad\text{2乗}
\end{aligned}$$
> $$\begin{aligned}
	\text{例として}e_{p},e_{q},e_{n}\text{の順であれば以下の様にまとめられる}
\end{aligned}$$
> $$\begin{aligned}
	e_1 = e_{p} \\
	e_2 = e_{q} \\
	e_3 = e_{n}
\end{aligned}$$
> $$\begin{aligned}
	e_ie_j + e_je_i &= 2g_{ij} \\
	g &= \mathrm{diag}\left(1,-1,0\right)
\end{aligned}$$
> $$\begin{aligned}
	\mathrm{diag}\left(1,-1,0\right) = \begin{bmatrix}
		1 & 0 & 0 \\
		0 & -1 & 0 \\
		0 & 0 & 0
	\end{bmatrix}
\end{aligned}$$
>  

> # バイベクトル  
>> ## 基底のペア  
>> $$\begin{aligned}
	&\text{異なる基底の2つ1組の積をバイベクトルと呼び、} \\
	&\text{組み合わさる種類としては以下のとおり}
\end{aligned}$$
>> $$\begin{aligned}
	&\text{同種の基底の関係} \\
	&e_{p1}e_{p2} \\
	&e_{q1}e_{q2} \\
	&e_{n1}e_{n2} \\
	\\
	&\text{異種の基底の関係} \\
	&e_{p}e_{q} \\
	&e_{p}e_{n} \\
	&e_{q}e_{n}
\end{aligned}$$
>> $$\begin{aligned}
	\text{上記の反交換関係にあるもの}
\end{aligned}$$
>> $$\begin{aligned}
	&\text{同種の基底の関係} \\
	&e_{p2}e_{p1} \\
	&e_{q2}e_{q1} \\
	&e_{n2}e_{n1} \\
	\\
	&\text{異種の基底の関係} \\
	&e_{q}e_{p} \\
	&e_{n}e_{p} \\
	&e_{n}e_{q}
\end{aligned}$$
>  
>> ## バイベクトルの2乗  
>> $$\begin{aligned}
	\text{同種の基底の関係}
\end{aligned}$$
>> $$\begin{aligned}
	(e_{p1}e_{p2})^2 &= (e_{p1}e_{p2})(e_{p1}e_{p2}) \\
	&= e_{p1}(e_{p2}e_{p1})e_{p2} \\
	&= e_{p1}(-e_{p1}e_{p2})e_{p2} \\
	&= -e_{p1}e_{p1}e_{p2}e_{p2} \\
	&= -(e_{p1}e_{p1})(e_{p2}e_{p2}) \\
	&= -(1)(1) \\
	&= -1 \\
	(e_{q1}e_{q2})^2 &= (e_{q1}e_{q2})(e_{q1}e_{q2}) \\
	&= e_{q1}(e_{q2}e_{q1})e_{q2} \\
  	&= e_{q1}(-e_{q1}e_{q2})e_{q2} \\
	&= -e_{q1}e_{q1}e_{q2}e_{q2} \\
	&= -(e_{q1}e_{q1})(e_{q2}e_{q2}) \\
	&= -(-1)(-1) \\
	&= -1 \\
	(e_{n1}e_{n2})^2 &= (e_{n1}e_{n2})(e_{n1}e_{n2}) \\
	&= e_{n1}(e_{n2}e_{n1})e_{n2} \\
	&= e_{n1}(-e_{n1}e_{n2})e_{n2} \\
	&= -e_{n1}e_{n1}e_{n2}e_{n2} \\
	&= -(e_{n1}e_{n1})(e_{n2}e_{n2}) \\
	&= -(0)(0) \\
	&= 0
\end{aligned}$$
>> $$\begin{aligned}
	\text{異種の基底の関係}
\end{aligned}$$
>> $$\begin{aligned}
	(e_{p}e_{q})^2 &= (e_{p}e_{q})(e_{p}e_{q}) \\
	&= e_{p}(e_{q}e_{p})e_{q} \\
	&= e_{p}(-e_{p}e_{q})e_{q} \\
	&= -e_{p}e_{p}e_{q}e_{q} \\
	&= -(e_{p}e_{p})(e_{q}e_{q}) \\
	&= -(1)(-1) \\
	&= 1 \\
	(e_{p}e_{n})^2 &= (e_{p}e_{n})(e_{p}e_{n}) \\
	&= e_{p}(e_{n}e_{p})e_{n} \\
	&= e_{p}(-e_{p}e_{n})e_{n} \\
	&= -e_{p}e_{p}e_{n}e_{n} \\
	&= -(e_{p}e_{p})(e_{n}e_{n}) \\
	&= -(1)(0) \\
	&= 0 \\
	(e_{q}e_{n})^2 &= (e_{q}e_{n})(e_{q}e_{n}) \\
	&= e_{q}(e_{n}e_{q})e_{n} \\
	&= e_{q}(-e_{q}e_{n})e_{n} \\
	&= -e_{q}e_{q}e_{n}e_{n} \\
	&= -(e_{q}e_{q})(e_{n}e_{n}) \\
	&= -(-1)(0) \\
	&= 0
\end{aligned}$$
>> $$\begin{aligned}
	\text{反交換関係にある組は省略}
\end{aligned}$$
>> $$\begin{aligned}
	\text{2乗して}\pm{1}\text{になるものを}\mathrm{\widehat{B}}\text{(単位バイベクトル)}と表す \\
	\text{2乗して}0\text{になるものを}B^2 = 0\text{(ヌルバイベクトル)}と表す
\end{aligned}$$
>  
>> ## 【まとめ】  
>> |バイベクトル|反交換関係のバイベクトル|$2乗$|
>> |-|-|-|
>> |$(e_{p1}e_{p2})$, $(e_{q1}e_{q2})$|$(e_{p2}e_{p1})$, $(e_{q2}e_{q1})$|$\mathrm{\widehat{B}^2}=-1$|
>> |$(e_{p}e_{q})$|$(e_{q}e_{p})$|$\mathrm{\widehat{B}^2}=1$|
>> |$(e_{n1}e_{n2})$, $(e_{p}e_{n})$, $(e_{q}e_{n})$|$(e_{n2}e_{n1})$, $(e_{n}e_{p})$, $(e_{n}e_{q})$|$B^2=0$|
>  

> # バイベクトルの線形結合  
> $$\begin{aligned}
	c_1,c_2,c_3\quad\text{係数(スカラー)} \\
	B_1,B_2,B_3\quad\text{バイベクトル}
\end{aligned}$$
> $$\begin{aligned}
	(c_1B_1 + c_2B_2 + c_3B_3)^2\text{について線形結合の様子を確認する}
\end{aligned}$$
> $$\begin{aligned}
	(c_1B_1 + c_2B_2 + c_3B_3)^2 &= (c_1B_1 + c_2B_2 + c_3B_3)(c_1B_1 + c_2B_2 + c_3B_3)
\end{aligned}$$
> $$\begin{aligned}
	&= (c_1B_1)^2 + c_1B_1c_2B_2 + c_1B_1c_3B_3\quad\text{まずは素直に展開} \\
	&+ c_2B_2c_1B_1 + (c_2B_2)^2 + c_2B_2c_3B_3 \\
	&+ c_3B_3c_1B_1 + c_3B_3c_2B_2 + (c_3B_3)^2 \\
\end{aligned}$$
> $$\begin{aligned}
	&= (c_1B_1)^2 + c_1c_2B_1B_2 + c_1c_3B_1B_3\quad\text{係数は可換なので順序を整理} \\
	&+ (c_2B_2)^2 + c_1c_2B_2B_1 + c_2c_3B_2B_3 \\
	&+ (c_3B_3)^2 + c_1c_3B_3B_1 + c_2c_3B_3B_2 \\
\end{aligned}$$
> $$\begin{aligned}
	&= (c_1B_1)^2 + (c_2B_2)^2 + (c_3B_3)^2\quad\text{同じ係数を括る} \\
	&+ c_1c_2(B_1B_2 + B_2B_1) \\
	&+ c_1c_3(B_1B_3 + B_3B_1) \\
	&+ c_2c_3(B_2B_3 + B_3B_2)
\end{aligned}$$
> $$\begin{aligned}
	\text{この結果から定式化すると}
\end{aligned}$$
> $$\begin{aligned}
	\left(\sum_{i=1}^n c_iB_i\right)^2 = \sum_{i=1}^n\underbrace{{c_i}^2{B_i}^2}_{\text{対角項}}+\sum_{i<j}\underbrace{c_ic_j(B_iB_j + B_jB_i)}_{\text{交差項}}
\end{aligned}$$
> $$\begin{aligned}
	\text{対角項はスカラー、交差項について調べる}
\end{aligned}$$
>> ## 反交換子
>> $$\begin{aligned}
	\lbrace{B_i,B_j}\rbrace = B_iB_j + B_jB_i
\end{aligned}$$
>> $$\begin{aligned}
	\underbrace{c_ic_j\lbrace{B_i,B_j}\rbrace}_{\text{上記の交差項を反交換子を用いて記述した例}}
\end{aligned}$$
>> ## 交換子
>> $$\begin{aligned}
	\lbrack{B_i,B_j}\rbrack = B_iB_j - B_jB_i
\end{aligned}$$

> # バイベクトルの2乗がスカラーになる場合の指数写像  
> $$\begin{aligned}
	\text{指数関数}e^\theta\text{のマクローリン展開をまず示すと} \\
	e^\theta = \sum^\infty_{n=0}\frac{\theta^n}{n!} = 1 + \theta + \frac{\theta^2}{2!} + \frac{\theta^3}{3!} + \frac{\theta^4}{4!} + \frac{\theta^5}{5!} + \cdots
\end{aligned}$$
>  
>> ## バイベクトルの2乗 $(B^2<0)$ の場合  
>> $$\begin{aligned}
	\text{単位バイベクトルを作る}
\end{aligned}$$
>> $$\begin{aligned}
	&\omega = \|B\|_2 \\
	&\mathrm{\widehat{B}} = \frac{B}{\omega}\quad\left(\mathrm{\widehat{B}^2} = -1\right) \\
	&x = \omega\theta
\end{aligned}$$
>> $$\begin{aligned}
	\text{累乗を確認して周期性に注目}
\end{aligned}$$
>> $$\begin{aligned}
	&\mathrm{\widehat{B}^2} = -1 \\
	&\mathrm{\widehat{B}^3} = \mathrm{\widehat{B}^2}\mathrm{\widehat{B}} = -\mathrm{\widehat{B}} \\
	&\mathrm{\widehat{B}^4} = \left(\mathrm{\widehat{B}^2}\right)^2 = 1 \\
	&\mathrm{\widehat{B}^5} = \mathrm{\widehat{B}^4}\mathrm{\widehat{B}} = \mathrm{\widehat{B}}
\end{aligned}$$
>> $$\begin{aligned}
	\text{ここで指数関数のマクローリン展開の}\theta\text{に}x\mathrm{\widehat{B}}\text{を代入すると}
\end{aligned}$$
>> $$\begin{aligned}
	e^{x\mathrm{\widehat{B}}} &= 1 &+& x\mathrm{\widehat{B}} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^2}{2!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^3}{3!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^4}{4!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^5}{5!} &+& \cdots \\
	&= 1 &+& x\mathrm{\widehat{B}} &-& \frac{x^2}{2!} &-& \frac{x^3}{3!}\mathrm{\widehat{B}} &+& \frac{x^4}{4!} &+& \frac{x^5}{5!}\mathrm{\widehat{B}} &-& \cdots
\end{aligned}$$
>> $$\begin{aligned}
	&= \left(1 - \frac{x^2}{2!} + \frac{x^4}{4!} - \cdots\right) + \mathrm{\widehat{B}}\left(x - \frac{x^3}{3!} + \frac{x^5}{5!} - \cdots\right)
\end{aligned}$$
>> $$\begin{aligned}
	\text{となり、}e^{i\theta}\text{のマクローリン展開と同じ構造となる}
\end{aligned}$$
>> $$\begin{aligned}
	e^{i\theta} = \left(1 - \frac{\theta^2}{2!} + \frac{\theta^4}{4!} - \cdots\right) + i\left(\theta - \frac{\theta^3}{3!} + \frac{\theta^5}{5!} - \cdots\right)
\end{aligned}$$
>> $$\begin{aligned}
	\therefore e^{x\mathrm{\widehat{B}}} &= \cos{x} + \mathrm{\widehat{B}}\sin{x}\quad\left(x = \omega\theta, \omega = \|B\|_2\right) \\
\end{aligned}$$
>> $$\begin{aligned}
	&\mathrm{\widehat{B}}\sin{x}\text{について}\left(x = \omega\theta, \mathrm{\widehat{B}} = \frac{B}{\omega}\right)\text{なので} \\
	&\mathrm{\widehat{B}}\sin{x} \Longrightarrow B\theta\frac{\sin{\omega\theta}}{\omega\theta}\text{と変形できる}
\end{aligned}$$
>> $$\begin{aligned}
	\therefore e^{\theta B} = \cos{\omega\theta} + B\theta\frac{\sin{\omega\theta}}{\omega\theta},\omega = \|B\|_2, \mathrm{\widehat{B}} = \frac{B}{\omega}\left(\mathrm{\widehat{B}^2} = -1\right)
\end{aligned}$$
>  
>> ## バイベクトルの2乗 $(B^2>0)$ の場合  
>> $$\begin{aligned}
	\text{単位バイベクトルを作る}
\end{aligned}$$
>> $$\begin{aligned}
	&\omega = \|B\|_2 \\
	&\mathrm{\widehat{B}} = \frac{B}{\omega}\quad\left(\mathrm{\widehat{B}^2} = 1\right) \\
	&x = \omega\theta
\end{aligned}$$
>> $$\begin{aligned}
	\text{累乗を確認すると符号の変化はなく、偶数乗が}1\text{、奇数乗が}\mathrm{\widehat{B}}\text{となる}
\end{aligned}$$
>> $$\begin{aligned}
	&\mathrm{\widehat{B}^2} = 1 \\
	&\mathrm{\widehat{B}^3} = \mathrm{\widehat{B}^2}\mathrm{\widehat{B}} = \mathrm{\widehat{B}} \\
	&\mathrm{\widehat{B}^4} = \left(\mathrm{\widehat{B}^2}\right)^2 = 1 \\
	&\mathrm{\widehat{B}^5} = \mathrm{\widehat{B}^4}\mathrm{\widehat{B}} = \mathrm{\widehat{B}}
\end{aligned}$$
>> $$\begin{aligned}
	\text{ここで指数関数のマクローリン展開の}\theta\text{に}x\mathrm{\widehat{B}}\text{を代入すると}
\end{aligned}$$
>> $$\begin{aligned}
	e^{x\mathrm{\widehat{B}}} &= 1 &+& x\mathrm{\widehat{B}} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^2}{2!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^3}{3!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^4}{4!} &+& \frac{\left(x\mathrm{\widehat{B}}\right)^5}{5!} &+& \cdots \\
	&= 1 &+& x\mathrm{\widehat{B}} &+& \frac{x^2}{2!} &+& \frac{x^3}{3!}\mathrm{\widehat{B}} &+& \frac{x^4}{4!} &+& \frac{x^5}{5!}\mathrm{\widehat{B}} &+& \cdots
\end{aligned}$$
>> $$\begin{aligned}
	&= \left(1 + \frac{x^2}{2!} + \frac{x^4}{4!} + \cdots\right) + \mathrm{\widehat{B}}\left(x + \frac{x^3}{3!} + \frac{x^5}{5!} + \cdots\right)
\end{aligned}$$
>> $$\begin{aligned}
	\text{となり、双曲線関数のマクローリン展開と同じ構造となる}
\end{aligned}$$
>> $$\begin{aligned}
	\cosh{\theta} &= 1 + \frac{\theta^2}{2!} + \frac{\theta^4}{4!} + \cdots \\
	\sinh{\theta} &= \theta + \frac{\theta^3}{3!} + \frac{\theta^5}{5!} + \cdots
\end{aligned}$$
>> $$\begin{aligned}
	\therefore e^{x\mathrm{\widehat{B}}} = \cosh{x} + \mathrm{\widehat{B}}\sinh{x}\quad\left(x = \omega\theta,\omega = \|B\|_2\right) \\
\end{aligned}$$
>> $$\begin{aligned}
	&\mathrm{\widehat{B}}\sinh{x}\text{について}\left(x = \omega\theta,\mathrm{\widehat{B}} = \frac{B}{\omega}\right)\text{なので} \\
	&\mathrm{\widehat{B}}\sinh{x} \Longrightarrow B\theta\frac{\sinh{\omega\theta}}{\omega\theta}\text{と変形できる} \\
	&\therefore e^{\theta B} = \cosh{\omega\theta} + B\theta\frac{\sinh{\omega\theta}}{\omega\theta}\quad\omega = \|B\|_2,\mathrm{\widehat{B}} = \frac{B}{\omega}\left(\mathrm{\widehat{B}^2} = 1\right)
\end{aligned}$$
>  
>> ## バイベクトルの2乗 $(B^2=0)$ の場合  
>> $$\begin{aligned}
	\text{2乗が0になるため、2乗以降の項はすべて0となる}
\end{aligned}$$
>> $$\begin{aligned}
	&B^2 = 0 \\
	&B^3 = {B^2}B = 0 \\
	&B^4 = \left(B^2\right)^2 = 0 \\
	&B^5 = {B^4}B = 0
\end{aligned}$$
>> $$\begin{aligned}
	\text{指数関数のマクローリン展開の}\theta\text{に}\theta B\text{を代入すると}
\end{aligned}$$
>> $$\begin{aligned}
	e^{\theta B} &= 1 &+& \theta B &+& \frac{\left(\theta B\right)^2}{2!} &+& \frac{\left(\theta B\right)^3}{3!} &+& \frac{\left(\theta B\right)^4}{4!} &+& \frac{\left(\theta B\right)^5}{5!} &+& \cdots \\
	&= 1 &+& \theta B &+& 0 &+& 0 &+& 0 &+& 0 &+&\cdots \\
	&= 1 &+& \theta B
\end{aligned}$$
>  
>> ## 【まとめ】  
>> |$B^2$|指数写像||
>> |-|-|-|
>> |$<0$|$e^{\theta B} = \cos{\omega\theta} + B\theta\dfrac{\sin{\omega\theta}}{\omega\theta}$|$\omega = \|B\|_2,\mathrm{\widehat{B}} = \dfrac{B}{\omega}\left(\mathrm{\widehat{B}^2} = -1\right)$|
>> |$=0$|$e^{\theta B} = 1 + \theta B$|
>> |$>0$|$e^{\theta B} = \cosh{\omega\theta} + B\theta\dfrac{\sinh{\omega\theta}}{\omega\theta}$|$\omega = \|B\|_2,\mathrm{\widehat{B}} = \dfrac{B}{\omega}\left(\mathrm{\widehat{B}^2} = 1\right)$|
>
