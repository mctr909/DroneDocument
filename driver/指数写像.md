## 計量と空間の設定
$$\begin{aligned}
	e_ie_j + e_je_i &= 2g_{ij} \\
	g &= \rm{diag}(-1,-1,-1) \\
	R &= r_s + r_{2,3}e_2e_3 + r_{3,1}e_3e_1 + r_{1,2}e_1e_2 \\
	\widetilde{R} &= r_s - r_{2,3}e_2e_3 - r_{3,1}e_3e_1 - r_{1,2}e_1e_2 \\
	v_{world} &= R(v_{local})\widetilde{R}\quad\cdots\text{接空間・余接空間の写像関係} \\
	R_{new} &= R_{old}\Delta R_1\Delta R_2\cdots\Delta R_n\quad\cdots\text{ローターの合成順は右積}
\end{aligned}$$

## ローター積
$$\begin{aligned}
	R_{\rm{a}}R_{\rm{b}} &= (c_{\rm{a}} + s_{\rm{a}}\rm{A})(c_{\rm{b}} + s_{\rm{b}}\rm{B}) \\
	&= c_{\rm{a}}c_{\rm{b}} + c_{\rm{a}}s_{\rm{b}}\rm{B} + s_{\rm{a}}c_{\rm{b}}\rm{A} + s_{\rm{a}}s_{\rm{b}}\rm{A}\rm{B}
\end{aligned}$$
$$\begin{aligned}
	&I=e_1e_2e_3\text{において}\quad Ie_1=-e_2e_3, Ie_2=-e_3e_1, Ie_3=-e_1e_2\text{となるので}
\end{aligned}$$
$$\begin{aligned}
	\rm{A}\rm{B} &= -\rm{A}\cdot_E\rm{B} - I(\rm{A}\times\rm{B}) \\
	&= -\rm{A}_{2,3}\rm{B}_{2,3} - \rm{A}_{3,1}\rm{B}_{3,1} - \rm{A}_{1,2}\rm{B}_{1,2} \\
	&+ (\rm{A}_{3,1}\rm{B}_{1,2} - \rm{A}_{1,2}\rm{B}_{3,1})e_2e_3 \\
	&+ (\rm{A}_{1,2}\rm{B}_{2,3} - \rm{A}_{2,3}\rm{B}_{1,2})e_3e_1 \\
	&+ (\rm{A}_{2,3}\rm{B}_{3,1} - \rm{A}_{3,1}\rm{B}_{2,3})e_1e_2
\end{aligned}$$
$$\begin{aligned}
	\rm{S} &= -(\rm{A}_{2,3}\rm{B}_{2,3} + \rm{A}_{3,1}\rm{B}_{3,1} + \rm{A}_{1,2}\rm{B}_{1,2}) \\
	\rm{W}_{2,3} &= \rm{A}_{3,1}\rm{B}_{1,2} - \rm{A}_{1,2}\rm{B}_{3,1} \\
	\rm{W}_{3,1} &= \rm{A}_{1,2}\rm{B}_{2,3} - \rm{A}_{2,3}\rm{B}_{1,2} \\
	\rm{W}_{1,2} &= \rm{A}_{2,3}\rm{B}_{3,1} - \rm{A}_{3,1}\rm{B}_{2,3}
\end{aligned}$$
$$\begin{aligned}
	R_{\rm{a}}R_{\rm{b}} &= c_{\rm{a}}c_{\rm{b}} + s_{\rm{a}}s_{\rm{b}}\rm{S} \\
	&+ \left(c_{\rm{a}}s_{\rm{b}}\rm{B}_{2,3} + s_{\rm{a}}c_{\rm{b}}\rm{A}_{2,3} + s_{\rm{a}}s_{\rm{b}}\rm{W}_{2,3}\right)e_2e_3 \\
	&+ \left(c_{\rm{a}}s_{\rm{b}}\rm{B}_{3,1} + s_{\rm{a}}c_{\rm{b}}\rm{A}_{3,1} + s_{\rm{a}}s_{\rm{b}}\rm{W}_{3,1}\right)e_3e_1 \\
	&+ \left(c_{\rm{a}}s_{\rm{b}}\rm{B}_{1,2} + s_{\rm{a}}c_{\rm{b}}\rm{A}_{1,2} + s_{\rm{a}}s_{\rm{b}}\rm{W}_{1,2}\right)e_1e_2
\end{aligned}$$

## 現在の姿勢における鉛直方向と加速度センサー $(a_{\rm{sens}})$ による比較ベクトル $(v_{\rm{ref}}, v_{\rm{target}})$ の取得
$$\begin{aligned}
	v_{\rm{target}} &= \frac{a_{\rm{sens}}}{\|a_{\rm{sens}}\|_2}\quad\cdots\text{正規化加速度センサー値} \\
	v_{\rm{ref}} &= \widetilde{R}(e_3)R\quad\cdots\text{現在の姿勢における鉛直方向}
\end{aligned}$$

## 方位センサー $(m_{\rm{sens}})$ を基にした現在の姿勢における水平面と正規化方位センサー値による比較ベクトル $(v_{\rm{ref}}, v_{\rm{target}})$ の取得
$$\begin{aligned}
	v_{\rm{target}} &= \frac{m_{\rm{sens}}}{\|m_{\rm{sens}}\|_2}\quad\cdots\text{正規化方位センサー値} \\
	v_w &= R(v_{\rm{target}})\widetilde{R}\quad\cdots\text{プッシュフォワード} \\
	v_{wh} &= \frac{{v_w}_1e_1 + {v_w}_2e_2}{\|{v_w}_1e_1 + {v_w}_2e_2\|_2}\quad\cdots\text{水平面を切り取る} \\
	v_{\rm{ref}} &= \widetilde{R}(v_{wh})R\quad\cdots\text{プルバック, 方位センサーを基にした現在の姿勢における水平面}
\end{aligned}$$

## 比較ベクトル $(v_{\rm{ref}}, v_{\rm{target}})$ から補正双ベクトル $(\Gamma)$ を作成
$$\begin{aligned}
	a &= {v_{\rm{target}}}_1e_1 + {v_{\rm{target}}}_2e_2 + {v_{\rm{target}}}_3e_3 \\
	b &= {v_{\rm{ref}}}_1e_1 + {v_{\rm{ref}}}_2e_2 + {v_{\rm{ref}}}_3e_3
\end{aligned}$$
$$\begin{aligned}
	\Gamma &= a\wedge b \\
	&= (a_2b_3 - a_3b_2)e_2e_3 \\
	&+ (a_3b_1 - a_1b_3)e_3e_1 \\
	&+ (a_1b_2 - a_2b_1)e_1e_2
\end{aligned}$$
$$\begin{aligned}
	&\text{厳密な誤差回転量は以下のとおりであるが小角近似を用いる}
\end{aligned}$$
$$\begin{aligned}
	\phi &= \rm{atan2}(\|a\wedge b\|_2, a\cdot_E b) \\
	B &= \frac{a\wedge b}{\|a\wedge b\|_2} \\
	&e^{\textstyle{B}\dfrac{\phi}{2}}
\end{aligned}$$

## 補正双ベクトル $(\Gamma)$ と補正速度 $(\Gamma_{\rm scale})$ による補正ローター $(\Delta R_{\textstyle\gamma})$ の作成
$$\begin{aligned}
	\Gamma^2 &= -(\Gamma_{2,3})^2 - (\Gamma_{3,1})^2 - (\Gamma_{1,2})^2 \\
	&\cdots\text{交差項が消える過程は別資料を参照} \\
	\gamma^2 &= (\Gamma_{2,3})^2 + (\Gamma_{3,1})^2 + (\Gamma_{1,2})^2 \\
	&\therefore \Gamma^2 = -\gamma^2
\end{aligned}$$
$$\begin{aligned}
	&x = \Gamma_{{\rm scale}}\gamma\theta,\quad\theta = \frac{\Delta t}{2} \\
	&\Delta R_{\textstyle\gamma} = e^{\textstyle\frac{\Gamma}{\gamma}x} = \cos{x} + \frac{\Gamma}{\gamma}\sin{x} \\
	&\frac{\Gamma}{\gamma}\sin{x} \Longrightarrow \Gamma_{\rm scale}\theta\Gamma\frac{\sin{x}}{x}
\end{aligned}$$
$$\begin{aligned}
	c_{\textstyle\gamma} = \cos{x} \approx \frac{144-60x^{2}+x^{4}}{144+12x^{2}+x^{4}},\quad s_{\textstyle\gamma} = \Gamma_{\rm scale}\theta\frac{\sin{x}}{x} \approx \Gamma_{\rm scale}\theta\frac{144-12x^{2}}{144+12x^{2}+x^{4}}
\end{aligned}$$
$$\begin{aligned}
	&\Delta R_{\textstyle\gamma} \approx c_{\textstyle\gamma} + s_{\textstyle\gamma}\left(\Gamma_{2,3}e_2e_3 + \Gamma_{3,1}e_3e_1 + \Gamma_{1,2}e_1e_2\right)
\end{aligned}$$

## 補正双ベクトル $(\Gamma)$ を蓄積してドリフト補正値 $(E)$ を取得
$$\begin{aligned}
	&E = \int 2\pi f_c(\Gamma-E)dt \Longrightarrow \frac{dE}{dt} = 2\pi f_c(\Gamma-E)
\end{aligned}$$
$$\begin{aligned}
	&E_{n+1} = E_n + E_{\rm{damp}}(\Gamma_n - E_n) \\
	&E_{\rm{damp}} = 1-e^{-2\pi f_c \Delta t} \\
\end{aligned}$$
$$\begin{aligned}
	&x = 2\pi f_c \Delta t \\
	&e^{-x} \approx \frac{12+x^2-6x}{12+x^2+6x} \\
	&E_{\rm{damp}} \approx 1-\frac{12+x^2-6x}{12+x^2+6x} = \frac{12x}{12+x^2+6x}
\end{aligned}$$

## 角速度センサー $(w_{\rm{sens}})$ とドリフト補正量 $(2\pi{K_i}E)$ によるローター $(\Delta R_{\textstyle\omega})$ の作成
$$\begin{aligned}
	w &= w_{\rm{sens}} + 2\pi{K_i}E \\
	\Omega &= w_{2,3}e_2e_3 + w_{3,1}e_3e_1 + w_{1,2}e_1e_2
\end{aligned}$$
$$\begin{aligned}
	\Omega^2 &= -(w_{2,3})^2 - (w_{3,1})^2 - (w_{1,2})^2 \\
	&\cdots\text{交差項が消える過程は別資料を参照} \\
	\omega^2 &= (w_{2,3})^2 + (w_{3,1})^2 + (w_{1,2})^2 \\
	&\therefore\Omega^2 = -\omega^2
\end{aligned}$$
$$\begin{aligned}
	&x = \omega\theta,\quad\theta = \frac{\Delta t}{2} \\
	&\Delta R_{\textstyle\omega} = e^{\textstyle\frac{\Omega}{\omega}x} = \cos{x} + \frac{\Omega}{\omega}\sin{x} \\
	&\frac{\Omega}{\omega}\sin{x} \Longrightarrow \theta\Omega\frac{\sin{x}}{x}
\end{aligned}$$
$$\begin{aligned}
	c_{\textstyle\omega} = \cos{x} \approx \frac{144-60x^{2}+x^{4}}{144+12x^{2}+x^{4}},\quad s_{\textstyle\omega} = \theta\frac{\sin{x}}{x} \approx \theta\frac{144-12x^{2}}{144+12x^{2}+x^{4}}
\end{aligned}$$
$$\begin{aligned}
	\Delta R_{\textstyle\omega} \approx c_{\textstyle\omega} + s_{\textstyle\omega}\left(w_{2,3}e_2e_3 + w_{3,1}e_3e_1 + w_{1,2}e_1e_2\right)
\end{aligned}$$
